                if (!soundEnabled) return;
                try {
                    const ctx = getAudioContext();
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    const now = ctx.currentTime;
                    if (type === 'correct') {
                        osc.type = 'sine';
                        osc.frequency.setValueAtTime(523.25, now);
                        osc.frequency.setValueAtTime(659.25, now + 0.1);
                        gain.gain.setValueAtTime(0.3, now);
                        gain.gain.linearRampToValueAtTime(0, now + 0.25);
                        osc.start(now);
                        osc.stop(now + 0.25);
                    } else if (type === 'incorrect') {
                        osc.type = 'square';
                        osc.frequency.setValueAtTime(200, now);
                        gain.gain.setValueAtTime(0.2, now);
                        gain.gain.linearRampToValueAtTime(0, now + 0.3);
                        osc.start(now);
                        osc.stop(now + 0.3);
                    }
                } catch (e) {}
            }

            function getRandomLetter(exclude = []) {
                const available = LATIN_LETTERS.filter(l => !exclude.includes(l));
                if (available.length === 0) return 'X';
                return available[Math.floor(Math.random() * available.length)];

            function generateLineNames() {

            // ════════ الرسم ════════
            function drawFigure() {
                const w = canvas.width = 850;
                const h = canvas.height = 480;
                ctx.clearRect(0, 0, w, h);

                const style = getComputedStyle(document.documentElement);
                const lineColor = style.getPropertyValue('--line-color').trim() || '#1f5e8c';
                const transColor = style.getPropertyValue('--trans-color').trim() || '#c0392b';
                const textColor = style.getPropertyValue('--text-color-canvas').trim() || '#0b3b4f';
                const parallelColor = style.getPropertyValue('--parallel-sign-color').trim() || '#1f5e8c';

                ctx.save();
                const cx = w / 2;
                const cy = h / 2;
                ctx.translate(cx, cy);
                if (flipSign === -1) ctx.scale(-1, 1);
                ctx.rotate(rotationAngle);

                const xMin = -170, xMax = 170;
                const y1 = -40, y2 = 40;

                const l1Start = { x: xMin, y: y1 };
                const l1End = { x: xMax, y: y1 };
                const l2Start = { x: xMin, y: y2 };
                const l2End = { x: xMax, y: y2 };
                const transStart = { x: xMin + 40 + Math.random() * 30, y: y1 - 30 };
                const transEnd = { x: xMax - 40 - Math.random() * 30, y: y2 + 30 };

                function drawLine(p1, p2, color, width) {
                    ctx.beginPath();
                    ctx.moveTo(p1.x, p1.y);
                    ctx.lineTo(p2.x, p2.y);
                    ctx.strokeStyle = color;
                    ctx.lineWidth = width;
                    ctx.stroke();
                }

                drawLine(l1Start, l1End, lineColor, 3.5);
                drawLine(l2Start, l2End, lineColor, 3.5);
                drawLine(transStart, transEnd, transColor, 3.2);

                function getTransformedPoint(px, py) {
                    let x = px, y = py;
                    const cos = Math.cos(rotationAngle);
                    const sin = Math.sin(rotationAngle);
                    let xr = x * cos - y * sin;
                    let yr = x * sin + y * cos;
                    xr *= flipSign;
                    return { x: xr + cx, y: yr + cy };
                }

                ctx.restore();
                ctx.save();
                ctx.setTransform(1, 0, 0, 1, 0, 0);

                ctx.font = 'bold 20px "Segoe UI", "Arial", sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';

                function drawTextAt(px, py, text, color, offsetX = 0, offsetY = 0) {
                    const pos = getTransformedPoint(px, py);
                    ctx.fillStyle = color;
                    ctx.fillText(text, pos.x + offsetX, pos.y + offsetY);
                }

                drawTextAt(l1Start.x - 30, l1Start.y - 10, line1, textColor);
                drawTextAt(l2Start.x - 30, l2Start.y + 12, line2, textColor);
                const midTrans = {
                    x: (transStart.x + transEnd.x) / 2,
                    y: (transStart.y + transEnd.y) / 2
                };
                drawTextAt(midTrans.x + 20, midTrans.y - 8, transversal, transColor);

                ctx.font = '18px "Segoe UI"';
                ctx.fillStyle = parallelColor;
                const para1Pos = getTransformedPoint(l1End.x - 50, l1End.y - 12);
                const para2Pos = getTransformedPoint(l2End.x - 50, l2End.y - 12);
                ctx.fillText('//', para1Pos.x, para1Pos.y);
                ctx.fillText('//', para2Pos.x, para2Pos.y);

                ctx.font = '14px "Segoe UI"';
                ctx.fillStyle = textColor;
                const textPos = getTransformedPoint(l1End.x - 120, l1End.y - 20);
                ctx.fillText(t('parallel_label') || 'متوازيان', textPos.x, textPos.y);

                ctx.restore();
            }

                checkBtn.disabled = true;
                setTimeout(startNewFullRound, 2200);

            function handleCheck() {
                if (!gameActive) { messageEl.innerHTML = t('round_restart'); return; }
                if (attempts >= 10) { finishCurrentRound(); return; }

                const ansLine2 = (inputRefs.line2?.value || '').trim().toUpperCase();
                const ansTrans = (inputRefs.trans?.value || '').trim().toUpperCase();

                if (!ansLine2 || !ansTrans) {
                    messageEl.innerHTML = t('fill_all'); messageEl.style.color = 'var(--accent3)'; return;
                }
                if (!LATIN_LETTERS.includes(ansLine2) || !LATIN_LETTERS.includes(ansTrans)) {
                    messageEl.innerHTML = t('invalid_letters'); messageEl.style.color = 'var(--accent3)'; return;
                }

                const isCorrect = (ansLine2 === line2 && ansTrans === transversal);
                attempts++;

                if (isCorrect) {
                    score++;
                    messageEl.innerHTML = t('correct'); messageEl.style.color = 'var(--accent2)';
                    playSound('correct');
                } else {
                    messageEl.innerHTML = t('wrong', { line2, transversal });
                    messageEl.style.color = 'var(--accent3)';
                    playSound('incorrect');
                }

                updateStatsUI();
                if (attempts >= 10) { finishCurrentRound(); return; }
                generateNewQuestion();
                setTimeout(() => inputRefs.line2?.focus(), 40);
            }

                    if (messageEl.innerHTML === t('figure_changed')) messageEl.innerHTML = '';
                }, 1500);

            if (!groups[ex.section]) {
                groups[ex.section] = {
                    icon: ex.sectionIcon || '📝',
                    exercises: [],
                };

            groups[ex.section].exercises.push(ex);
        });
        const orderedSections = [];
        const seenSections = new Set();
        exercises.forEach(ex => {
            if (groups[ex.section] && !seenSections.has(ex.section)) {
                seenSections.add(ex.section);
                orderedSections.push(ex.section);
            }
        });
        Object.keys(groups).forEach(sec => {
            if (!seenSections.has(sec)) orderedSections.push(sec);
        });
        return { groups, orderedSections };

                <div class="empty-state">
                    <span class="empty-icon">🔍</span>
                    <p>${t('empty_state_text')}</p>
                    <p style="font-size:0.85rem;margin-top:6px;">${t('empty_state_sub')}</p>
                </div>`;
            return;
        }

        let html = '';
        orderedSections.forEach(sectionName => {
            const group = groups[sectionName];
            if (!group || group.exercises.length === 0) return;
            const isCollapsed = collapsedSections.has(sectionName);
            const cardsHTML = group.exercises.map(ex => renderCard(ex)).join('');
            html += `
            <div class="exercise-section${isCollapsed ? ' collapsed' : ''}" data-section="${escapeHTML(sectionName)}">
                <div class="section-header" data-section-name="${escapeHTML(sectionName)}">
                    <div class="section-title-area">
                        <span class="section-icon">${group.icon}</span>
                        <span>${escapeHTML(sectionName)}</span>
                        <span class="section-count">${t('section_count', { count: group.exercises.length })}</span>
                    </div>
                    <span class="section-toggle-icon">▼</span>
                </div>
                <div class="section-body">${cardsHTML}</div>
            </div>`;
        });
        mainContent.innerHTML = html;

        bindCardEvents();
        bindSectionToggleEvents();
    }

    function bindCardEvents() {
        mainContent.querySelectorAll('.exercise-card').forEach(card => {
            card.addEventListener('click', function(e) {
                if (e.target.classList.contains('card-tag')) {
                    e.stopPropagation();
                    const tag = e.target.dataset.tag;
                    if (tag) {
                        activeTags.clear();
                        activeTags.add(tag);
                        renderTagsBar();
                        renderAllSections();
                        updateStats();
                        window.scrollTo({ top: 0, behavior: 'smooth' });
                    }
                    return;
                }
                if (e.target.closest('[data-action="open"]')) {
                    e.stopPropagation();
                }
                const link = card.dataset.link;
                if (link) {
                    window.open(link, '_blank', 'noopener,noreferrer');
                }
            });

            card.addEventListener('dblclick', function(e) {
                e.preventDefault();
                const link = card.dataset.link;
                const name = card.dataset.name;
                if (link) {
                    openPreview(link, name);
                }
            });
        });

        mainContent.querySelectorAll('[data-action="open"]').forEach(btn => {
            btn.addEventListener('click', function(e) {

            });

    // ───── ربط أحداث طيّ الأقسام ─────
    function bindSectionToggleEvents() {
        mainContent.querySelectorAll('.section-header').forEach(header => {
            header.addEventListener('click', function(e) {
                e.stopPropagation();
                const sectionName = this.dataset.sectionName;
                const sectionEl = this.closest('.exercise-section');
                if (sectionEl) {
                    const isNowCollapsed = !sectionEl.classList.contains('collapsed');
                    if (isNowCollapsed) {
                        sectionEl.classList.add('collapsed');
                        collapsedSections.add(sectionName);
                    } else {
                        sectionEl.classList.remove('collapsed');
                        collapsedSections.delete(sectionName);
                    }
                    saveCollapsedState();
                }
            });
        });
    }

    function saveCollapsedState() {
        try {

        } catch (e) { /* ignore */ }
    }

    function loadCollapsedState() {
        try {
            const saved = localStorage.getItem('dashboard_collapsed_sections');
            if (saved) {
                const arr = JSON.parse(saved);
                collapsedSections = new Set(arr);
            }
        } catch (e) {
            collapsedSections = new Set();
        }
    }

        const current = htmlElement.getAttribute('data-theme');
        applyTheme(current === 'dark' ? 'light' : 'dark');
    }

    function loadTheme() {
        try {
            const saved = localStorage.getItem('dashboard_theme');
            if (saved === 'dark' || saved === 'light') {
                applyTheme(saved);
                return;
            }
        } catch (e) { /* ignore */ }
        if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
            applyTheme('dark');
        } else {
            applyTheme('light');
        }
    }

            function randomLetter(exclude = []) {

                const p2 = randomLetter([l1, l2, trans, p1]);

                if (Math.random() < 0.5) {
                    line1 = l1; line2 = l2;
                } else {
                    line1 = l2; line2 = l1;

                aux1 = randomLetter(used); used.push(aux1);

            function drawFigure() {
                canvas.width = 850;
                canvas.height = 480;
                ctx.setTransform(1, 0, 0, 1, 0, 0);
                ctx.clearRect(0, 0, canvas.width, canvas.height);

                const w = canvas.width, h = canvas.height;
                ctx.save();
                ctx.translate(w/2, h/2);
                if (flipSign === -1) ctx.scale(-1, 1);
                ctx.rotate(rotationAngle);

                const xMin = -170, xMax = 170;
                const y1 = -40, y2 = 40;
                const line1Start = {x:xMin, y:y1}, line1End = {x:xMax, y:y1};
                const line2Start = {x:xMin, y:y2}, line2End = {x:xMax, y:y2};
                const seed = transRandomSeed;
                const transStart = {x: xMin+40+seed*30, y: y1-30};
                const transEnd = {x: xMax-40-seed*30, y: y2+30};

                function drawLine(p1,p2,color,width) {
                    ctx.beginPath(); ctx.moveTo(p1.x,p1.y); ctx.lineTo(p2.x,p2.y);
                    ctx.strokeStyle=color; ctx.lineWidth=width; ctx.stroke();
                }
                drawLine(line1Start,line1End,'#1f5e8c',3.5);
                drawLine(line2Start,line2End,'#1f5e8c',3.5);
                drawLine(transStart,transEnd,'#c0392b',3.2);

                function intersect(p1,p2,p3,p4) {
                    const x1=p1.x,y1=p1.y,x2=p2.x,y2=p2.y,x3=p3.x,y3=p3.y,x4=p4.x,y4=p4.y;
                    const d=(x1-x2)*(y3-y4)-(y1-y2)*(x3-x4);
                    if(Math.abs(d)<1e-10) return null;
                    const t=((x1-x3)*(y3-y4)-(y1-y3)*(x3-x4))/d;
                    return {x:x1+t*(x2-x1), y:y1+t*(y2-y1)};
                }
                const inter1 = intersect(line1Start,line1End,transStart,transEnd);
                const inter2 = intersect(line2Start,line2End,transStart,transEnd);

                const focusPoint = (chosenPointIndex === 0) ? inter1 : inter2;
                const focusLineStart = (chosenPointIndex === 0) ? line1Start : line2Start;
                const focusLineEnd = (chosenPointIndex === 0) ? line1End : line2End;

                function drawAngleArc(vertex,dir1,dir2,radius,color,alpha) {
                    const a1=Math.atan2(dir1.y,dir1.x), a2=Math.atan2(dir2.y,dir2.x);
                    ctx.beginPath(); ctx.moveTo(vertex.x,vertex.y);
                    ctx.arc(vertex.x,vertex.y,radius,a1,a2); ctx.lineTo(vertex.x,vertex.y);
                    ctx.fillStyle=color; ctx.globalAlpha=alpha; ctx.fill(); ctx.globalAlpha=1;
                    ctx.strokeStyle=color; ctx.lineWidth=1.8; ctx.stroke();
                }

                // الزاويتان المتقابلتان بالرأس
                const toLeft = {x: focusLineStart.x-focusPoint.x, y: focusLineStart.y-focusPoint.y};
                const toRight = {x: focusLineEnd.x-focusPoint.x, y: focusLineEnd.y-focusPoint.y};
                const toTransUp = {x: transStart.x-focusPoint.x, y: transStart.y-focusPoint.y};
                const toTransDown = {x: transEnd.x-focusPoint.x, y: transEnd.y-focusPoint.y};

                // الزاوية 1: بين المستقيم (جهة اليسار) والقاطع (جهة الأعلى) – برتقالي
                drawAngleArc(focusPoint, toLeft, toTransUp, 26, '#e06c2e', 0.32);
                // الزاوية 2: بين المستقيم (جهة اليمين) والقاطع (جهة الأسفل) – برتقالي
                drawAngleArc(focusPoint, toRight, toTransDown, 26, '#e06c2e', 0.32);

                ctx.restore();

                function getTransformed(x,y) {
                    const cos=Math.cos(rotationAngle), sin=Math.sin(rotationAngle);
                    let xr=x*cos - y*sin, yr=x*sin + y*cos;
                    xr*=flipSign;
                    return {x:xr+w/2, y:yr+h/2};
                }
                function drawTextAt(px,py,text,color,ox=0,oy=0) {
                    const p=getTransformed(px,py);
                    ctx.font='bold 19px "Segoe UI"'; ctx.textAlign='center'; ctx.textBaseline='middle';
                    ctx.fillStyle=color; ctx.fillText(text,p.x+ox,p.y+oy);
                }
                drawTextAt(line1Start.x-30,line1Start.y-10,line1,'#0b3b4f');
                drawTextAt(line2Start.x-30,line2Start.y+12,line2,'#0b3b4f');
                drawTextAt((transStart.x+transEnd.x)/2+20,(transStart.y+transEnd.y)/2-8,transversal,'#a4281c');
                drawTextAt(inter1.x,inter1.y,point1,'#1e4b6e',14,-22);
                drawTextAt(inter2.x,inter2.y,point2,'#1e4b6e',14,24);

                function drawPoint(px,py,letter,color) {
                    const p=getTransformed(px,py);
                    ctx.beginPath(); ctx.arc(p.x,p.y,6.5,0,2*Math.PI); ctx.fillStyle=color; ctx.fill();
                    ctx.font='bold 20px "Segoe UI"'; ctx.fillStyle=color; ctx.fillText(letter,p.x+14,p.y-7);
                }

                const factor = 0.88;
                const angleColor = '#d45a20';
                const fp = focusPoint;
                const aux1Pos = {x: fp.x + (focusLineStart.x - fp.x) * factor, y: fp.y + (focusLineStart.y - fp.y) * factor};
                const aux2Pos = {x: fp.x + (focusLineEnd.x - fp.x) * factor, y: fp.y + (focusLineEnd.y - fp.y) * factor};
                const aux3Pos = {x: fp.x + (transStart.x - fp.x) * factor, y: fp.y + (transStart.y - fp.y) * factor};
                const aux4Pos = {x: fp.x + (transEnd.x - fp.x) * factor, y: fp.y + (transEnd.y - fp.y) * factor};

                drawPoint(aux1Pos.x, aux1Pos.y, aux1, angleColor);
                drawPoint(aux2Pos.x, aux2Pos.y, aux2, angleColor);
                drawPoint(aux3Pos.x, aux3Pos.y, aux3, angleColor);
                drawPoint(aux4Pos.x, aux4Pos.y, aux4, angleColor);

                // تمييز النقطة المختارة بدائرة إضافية
                const chosenP = getTransformed(fp.x, fp.y);
                ctx.beginPath();
                ctx.arc(chosenP.x, chosenP.y, 14, 0, 2*Math.PI);
                ctx.strokeStyle = '#e06c2e';
                ctx.lineWidth = 3;
                ctx.setLineDash([6, 3]);
                ctx.stroke();
                ctx.setLineDash([]);
            }

            function handleCheck() {
                if (!gameActive) {
                    messageEl.innerHTML = t('round_restart');
                    return;
                }
                if (attempts >= 10) {
                    finishCurrentRound();
                    return;
                }

                const ans = {
                    line2: inputFields.line2.value.trim().toUpperCase(),
                    trans: inputFields.trans.value.trim().toUpperCase(),
                    pt1: inputFields.pt1.value.trim().toUpperCase(),
                    pt2: inputFields.pt2.value.trim().toUpperCase(),
                    ptChosen: inputFields.ptChosen.value.trim().toUpperCase(),
                    angle1: inputFields.angle1.value.trim().toUpperCase(),
                    angle2: inputFields.angle2.value.trim().toUpperCase()
                };

                if (!ans.line2 || !ans.trans || !ans.pt1 || !ans.pt2 || !ans.ptChosen || !ans.angle1 || !ans.angle2) {
                    messageEl.innerHTML = t('fill_all');
                    return;
                }
                for (let s of [ans.line2, ans.trans, ans.pt1, ans.pt2, ans.ptChosen]) {
                    if (s.length !== 1 || !LATIN_LETTERS.includes(s)) {
                        messageEl.innerHTML = t('invalid_single');
                        return;
                    }
                }
                for (let a of [ans.angle1, ans.angle2]) {
                    if (a.length !== 3 || [...a].some(ch => !LATIN_LETTERS.includes(ch))) {
                        messageEl.innerHTML = t('invalid_angle');
                        return;
                    }
                }

                const pointsForward = ans.pt1 === point1 && ans.pt2 === point2;
                const pointsReversed = ans.pt1 === point2 && ans.pt2 === point1;
                const pointsCorrect = pointsForward || pointsReversed;
                const mainCorrect = ans.line2 === line2 && ans.trans === transversal && pointsCorrect;
                const chosenCorrect = (ans.ptChosen === chosenPoint);

                const possible = [angle1Name, angle2Name];
                function isAngleOk(a) {
                    const rev = a[2]+a[1]+a[0];
                    return possible.includes(a) || possible.includes(rev);
                }
                const angle1Ok = isAngleOk(ans.angle1);
                const angle2Ok = isAngleOk(ans.angle2);
                const isCorrect = mainCorrect && chosenCorrect && angle1Ok && angle2Ok && (new Set([ans.angle1, ans.angle2]).size === 2);

                attempts++;
                if (isCorrect) {
                    score++;
                    messageEl.innerHTML = t('correct');
                    messageEl.style.color = 'var(--accent2)';
                    playSound('correct');
                } else {
                    messageEl.innerHTML = t('wrong', {
                        line2, trans: transversal, pt1: point1, pt2: point2,
                        ptChosen: chosenPoint, angle1: angle1Name, angle2: angle2Name
                    });
                    messageEl.style.color = 'var(--accent3)';
                    playSound('incorrect');
                }

                updateStatsUI();

                if (attempts === 10) {
                    finishCurrentRound();
                    return;
                }
                generateNewQuestion();
                setTimeout(() => inputFields.line2?.focus(), 30);
            }

            // ══════════════ ربط الأحداث ══════════════

            function playSound(type) {
                if (!soundEnabled) return;
                try {
                    const ctx = getAudioContext();
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    const now = ctx.currentTime;
                    if (type === 'correct') {
                        osc.type = 'sine';
                        osc.frequency.setValueAtTime(523.25, now);
                        osc.frequency.setValueAtTime(659.25, now + 0.1);
                        gain.gain.setValueAtTime(0.3, now);
                        gain.gain.linearRampToValueAtTime(0, now + 0.25);
                        osc.start(now);
                        osc.stop(now + 0.25);
                    } else if (type === 'incorrect') {
                        osc.type = 'square';
                        osc.frequency.setValueAtTime(200, now);
                        gain.gain.setValueAtTime(0.2, now);
                        gain.gain.linearRampToValueAtTime(0, now + 0.3);
                        osc.start(now);
                        osc.stop(now + 0.3);
                    }
                } catch (e) {}
            }

            function getRandomLetter(exclude = []) {
                const available = LATIN_LETTERS.filter(l => !exclude.includes(l));
                if (available.length === 0) return 'X';
                return available[Math.floor(Math.random() * available.length)];
            }

            function shuffleArray(arr) {
                const a = arr.slice();
                for (let i = a.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [a[i], a[j]] = [a[j], a[i]];
                }
                return a;
            }

            function generateBaseTrianglePoints(canvasW, canvasH) {
                const cx = canvasW / 2;
                const cy = canvasH / 2;
                const maxR = Math.min(canvasW, canvasH) * 0.42;
                const minR = maxR * 0.6;

                const angles = [];
                let a1 = Math.random() * Math.PI * 0.5 + 0.3;
                let a2 = a1 + Math.random() * (Math.PI * 0.9) + 0.8;
                let a3 = a2 + Math.random() * (Math.PI * 0.7) + 0.8;
                angles.push(a1, a2, a3);
                const shuffled = shuffleArray(angles);

                const points = shuffled.map(a => {
                    const r = minR + Math.random() * (maxR - minR);
                    return { x: cx + r * Math.cos(a), y: cy + r * Math.sin(a) };
                });

                const centroidX = points.reduce((s, p) => s + p.x, 0) / 3;
                const centroidY = points.reduce((s, p) => s + p.y, 0) / 3;
                points.forEach(p => { p.x += cx - centroidX; p.y += cy - centroidY; });
                return points;
            }

            function applyRigidTransform(points, canvasW, canvasH) {
                const cx = canvasW / 2;
                const cy = canvasH / 2;
                const angle = (Math.random() * 1.4 - 0.7);
                const cos = Math.cos(angle);
                const sin = Math.sin(angle);

                const centroidX = points.reduce((s, p) => s + p.x, 0) / 3;
                const centroidY = points.reduce((s, p) => s + p.y, 0) / 3;

                const transformed = points.map(p => {
                    const dx = p.x - centroidX;
                    const dy = p.y - centroidY;
                    const rx = dx * cos - dy * sin;
                    const ry = dx * sin + dy * cos;
                    return { x: cx + rx, y: cy + ry };
                });

                const shiftX = (Math.random() - 0.5) * canvasW * 0.25;
                const shiftY = (Math.random() - 0.5) * canvasH * 0.2;
                const margin = 30;
                transformed.forEach(p => {
                    p.x += shiftX;
                    p.y += shiftY;
                    p.x = Math.min(Math.max(p.x, margin), canvasW - margin);
                    p.y = Math.min(Math.max(p.y, margin), canvasH - margin);
                });
                return transformed;
            }

            function angleAtVertex(prev, vertex, next) {
                const v1 = { x: prev.x - vertex.x, y: prev.y - vertex.y };
                const v2 = { x: next.x - vertex.x, y: next.y - vertex.y };
                const dot = v1.x * v2.x + v1.y * v2.y;
                const cross = v1.x * v2.y - v1.y * v2.x;
                return Math.atan2(Math.abs(cross), dot);
            }

            function matchTrianglesByAngles(points1, points2, labels1, labels2) {
                const n = 3;
                const angles1 = [];
                const angles2 = [];
                for (let i = 0; i < n; i++) {
                    const prev1 = points1[(i + 2) % n],
                        curr1 = points1[i],
                        next1 = points1[(i + 1) % n];
                    angles1.push({ index: i, label: labels1[i], angle: angleAtVertex(prev1, curr1, next1) });
                    const prev2 = points2[(i + 2) % n],
                        curr2 = points2[i],
                        next2 = points2[(i + 1) % n];
                    angles2.push({ index: i, label: labels2[i], angle: angleAtVertex(prev2, curr2, next2) });
                }
                angles1.sort((a, b) => a.angle - b.angle);
                angles2.sort((a, b) => a.angle - b.angle);

                const mapping = [];
                const used2 = new Set();
                for (let i = 0; i < n; i++) {
                    let bestJ = -1,
                        bestDiff = Infinity;
                    for (let j = 0; j < n; j++) {
                        if (used2.has(j)) continue;
                        const diff = Math.abs(angles1[i].angle - angles2[j].angle);
                        if (diff < bestDiff) { bestDiff = diff;
                            bestJ = j; }
                    }
                    used2.add(bestJ);
                    mapping.push({
                        tri1Vertex: angles1[i].label,
                        tri1Index: angles1[i].index,
                        tri2Vertex: angles2[bestJ].label,
                        tri2Index: angles2[bestJ].index
                    });
                }
                return mapping.sort((a, b) => a.tri1Index - b.tri1Index);
            }

            }

            // ══════════════ الرسم ══════════════
            function drawHashMarks(ctx, p1, p2, isDouble) {
                const mx = (p1.x + p2.x) / 2;
                const my = (p1.y + p2.y) / 2;
                const dx = p2.x - p1.x;
                const dy = p2.y - p1.y;
                const len = Math.sqrt(dx * dx + dy * dy);
                if (len < 1) return;
                const ux = -dy / len;
                const uy = dx / len;
                const markHalf = 7;

                ctx.save();
                ctx.strokeStyle = getComputedStyle(document.documentElement).getPropertyValue('--side-color').trim() || '#2e7d32';
                ctx.lineWidth = 2.4;
                ctx.lineCap = 'round';

                if (isDouble) {
                    for (let t = -1; t <= 1; t += 2) {
                        const offset = t * 3.5;
                        ctx.beginPath();
                        ctx.moveTo(mx + ux * offset - ux * markHalf, my + uy * offset - uy * markHalf);
                        ctx.lineTo(mx + ux * offset + ux * markHalf, my + uy * offset + uy * markHalf);
                        ctx.stroke();
                    }
                } else {
                    ctx.beginPath();
                    ctx.moveTo(mx - ux * markHalf, my - uy * markHalf);
                    ctx.lineTo(mx + ux * markHalf, my + uy * markHalf);
                    ctx.stroke();
                }
                ctx.restore();
            }

            function drawTriangle(ctx, pts, labels, canvasW, canvasH, mapping, isTri1, chosenVIdx) {
                ctx.clearRect(0, 0, canvasW, canvasH);
                ctx.save();

                const style = getComputedStyle(document.documentElement);
                const angleColor = style.getPropertyValue('--angle-color').trim() || '#d9443a';
                const angleLight = style.getPropertyValue('--angle-light').trim() || 'rgba(217,68,58,0.22)';
                const sideColor = style.getPropertyValue('--side-color').trim() || '#2e7d32';
                const neutralColor = style.getPropertyValue('--neutral-angle').trim() || '#9e9e9e';
                const neutralLight = style.getPropertyValue('--neutral-light').trim() || 'rgba(158,158,158,0.12)';
                const sideStroke = style.getPropertyValue('--side-stroke').trim() || '#2c3e50';
                const vertexFill = style.getPropertyValue('--vertex-fill').trim() || '#1a1a2e';
                const vertexStroke = style.getPropertyValue('--vertex-stroke').trim() || '#ffffff';
                const labelColor = style.getPropertyValue('--label-color').trim() || '#0b2b3b';

                let angleVertexIdx;
                if (isTri1) {
                    angleVertexIdx = chosenVIdx;
                } else {
                    const mapped = mapping.find(m => m.tri1Index === chosenVIdx);
                    angleVertexIdx = mapped ? mapped.tri2Index : chosenVIdx;
                }

                const side1Idx = angleVertexIdx;
                const side2Idx = (angleVertexIdx + 2) % 3;

                // رسم الأضلاع
                for (let i = 0; i < 3; i++) {
                    const j = (i + 1) % 3;
                    const isHighlighted = (i === side1Idx || i === side2Idx);
                    ctx.beginPath();
                    ctx.moveTo(pts[i].x, pts[i].y);
                    ctx.lineTo(pts[j].x, pts[j].y);
                    if (isHighlighted) {
                        ctx.strokeStyle = sideColor;
                        ctx.lineWidth = 3.2;
                    } else {
                        ctx.strokeStyle = sideStroke;
                        ctx.lineWidth = 2.2;
                    }
                    ctx.stroke();

                    if (isHighlighted) {
                        const isDouble = (i === side2Idx);
                        drawHashMarks(ctx, pts[i], pts[j], isDouble);
                    }
                }

                // رسم أقواس الزوايا
                for (let i = 0; i < 3; i++) {
                    const prev = pts[(i + 2) % 3];
                    const curr = pts[i];
                    const next = pts[(i + 1) % 3];

                    const isHighlightedAngle = (i === angleVertexIdx);
                    const arcColor = isHighlightedAngle ? angleColor : neutralColor;
                    const arcFill = isHighlightedAngle ? angleLight : neutralLight;
                    const arcRadius = isHighlightedAngle ? 26 : 18;

                    const v1 = { x: prev.x - curr.x, y: prev.y - curr.y };
                    const v2 = { x: next.x - curr.x, y: next.y - curr.y };
                    const ang1 = Math.atan2(v1.y, v1.x);
                    const ang2 = Math.atan2(v2.y, v2.x);

                    ctx.beginPath();
                    ctx.moveTo(curr.x, curr.y);
                    ctx.arc(curr.x, curr.y, arcRadius, ang1, ang2);
                    ctx.lineTo(curr.x, curr.y);
                    ctx.fillStyle = arcFill;
                    ctx.fill();
                    ctx.strokeStyle = arcColor;
                    ctx.lineWidth = isHighlightedAngle ? 2.8 : 1.6;
                    ctx.stroke();

                    if (isHighlightedAngle) {
                        const midAngle = (ang1 + ang2) / 2;
                        const labelR = arcRadius + 16;
                        const lx = curr.x + labelR * Math.cos(midAngle);
                        const ly = curr.y + labelR * Math.sin(midAngle);
                        ctx.font = 'bold 12px "Segoe UI", "Arial"';
                        ctx.textAlign = 'center';
                        ctx.textBaseline = 'middle';
                        ctx.fillStyle = arcColor;
                        ctx.fillText('∠', lx, ly);
                    }
                }

                // رسم النقاط والحروف
                for (let i = 0; i < 3; i++) {
                    const p = pts[i];
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, 7, 0, 2 * Math.PI);
                    ctx.fillStyle = vertexFill;
                    ctx.fill();
                    ctx.strokeStyle = vertexStroke;
                    ctx.lineWidth = 2.2;
                    ctx.stroke();

                    const offsetX = 19;
                    const offsetY = -18;
                    ctx.font = 'bold 18px "Segoe UI", "Arial"';
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillStyle = labelColor;
                    ctx.fillText(labels[i], p.x + offsetX, p.y + offsetY);
                }

                ctx.restore();
            }

            function drawFigures() {
                drawTriangle(ctx1, currentTri1Pts, tri1Vertices, 420, 360, angleMapping, true, chosenVertexIndex);
                drawTriangle(ctx2, currentTri2Pts, tri2Vertices, 420, 360, angleMapping, false, chosenVertexIndex);
            }

            function generateNewQuestion() {
                const data = generateAllData();
                currentTri1Pts = data.pts1;
                currentTri2Pts = data.pts2;
                renderQuestion();
                drawFigures();
            }

            function updateStatsUI() {
                attemptsSpan.textContent = attempts;
                scoreSpan.textContent = score;
            }

            function startNewFullRound() {
                if (isSwitchingRound) return;
                isSwitchingRound = true;
                attempts = 0;
                score = 0;
                gameActive = true;
                updateStatsUI();
                generateNewQuestion();
                messageEl.innerHTML = '';
                messageEl.style.color = 'var(--text)';
                setInputsDisabled(false);
                checkBtn.disabled = false;
                isSwitchingRound = false;
                setTimeout(() => inputRefs.tri1?.focus(), 60);
            }

            function setInputsDisabled(disabled) {
                for (let key in inputRefs) {
                    if (inputRefs[key]) inputRefs[key].disabled = disabled;
                }
            }

            function finishCurrentRound() {
                if (!gameActive) return;
                gameActive = false;
                messageEl.innerHTML = `<span style="background:#e9b35f;color:#2d2b26;padding:8px 22px;border-radius:60px;display:inline-block;">🎯 ${t('round_over',{score})}</span>`;
                setInputsDisabled(true);
                checkBtn.disabled = true;
                setTimeout(startNewFullRound, 2200);
            }

            function getAngleNamesForVertex(idx, labels) {
                const prev = labels[(idx + 2) % 3];
                const curr = labels[idx];
                const next = labels[(idx + 1) % 3];
                return {
                    standard: `${prev}${curr}${next}`,
                    reversed: `${next}${curr}${prev}`,
                    vertex: curr
                };
            }

            function vertexFromAngle(input, labels) {
                for (let i = 0; i < 3; i++) {
                    const { standard, reversed, vertex } = getAngleNamesForVertex(i, labels);
                    if (input === standard || input === reversed) return vertex;
                }
                return null;
            }

            function sideMatches(input, labels, idxA, idxB) {
                const forward = labels[idxA] + labels[idxB];
                const backward = labels[idxB] + labels[idxA];
                return input === forward || input === backward;
            }

            function handleCheck() {
                if (!gameActive) {
                    messageEl.innerHTML = t('round_restart');
                    messageEl.style.color = 'var(--text)';
                    return;
                }
                if (attempts >= 10) {
                    finishCurrentRound();
                    return;
                }

                const ansTri1 = (inputRefs.tri1?.value || '').trim().toUpperCase();
                const ansTri2 = (inputRefs.tri2?.value || '').trim().toUpperCase();
                const ansAng1 = (inputRefs.ang1?.value || '').trim().toUpperCase();
                const ansAng2 = (inputRefs.ang2?.value || '').trim().toUpperCase();
                const ansSideA1 = (inputRefs.sideA1?.value || '').trim().toUpperCase();
                const ansSideA2 = (inputRefs.sideA2?.value || '').trim().toUpperCase();
                const ansSideB1 = (inputRefs.sideB1?.value || '').trim().toUpperCase();
                const ansSideB2 = (inputRefs.sideB2?.value || '').trim().toUpperCase();
                const ansTri1b = (inputRefs.tri1b?.value || '').trim().toUpperCase();
                const ansTri2b = (inputRefs.tri2b?.value || '').trim().toUpperCase();

                const allVals = [ansTri1, ansTri2, ansAng1, ansAng2, ansSideA1, ansSideA2, ansSideB1, ansSideB2,
                    ansTri1b, ansTri2b
                ];
                if (allVals.some(v => !v)) {
                    messageEl.innerHTML = t('fill_all');
                    messageEl.style.color = 'var(--accent3)';
                    return;
                }

                for (let s of [ansTri1, ansTri2, ansAng1, ansAng2, ansTri1b, ansTri2b]) {
                    if (s.length !== 3) {
                        messageEl.innerHTML = t('invalid_format');
                        messageEl.style.color = 'var(--accent3)';
                        return;
                    }
                }
                for (let s of [ansSideA1, ansSideA2, ansSideB1, ansSideB2]) {
                    if (s.length !== 2) {
                        messageEl.innerHTML = t('invalid_format');
                        messageEl.style.color = 'var(--accent3)';
                        return;
                    }
                }

                const allInputs = [ansTri1, ansTri2, ansAng1, ansAng2, ansTri1b, ansTri2b, ansSideA1, ansSideA2,
                    ansSideB1, ansSideB2
                ];
                for (let str of allInputs) {
                    for (let ch of str) {
                        if (!LATIN_LETTERS.includes(ch)) {
                            messageEl.innerHTML = t('invalid_letters');
                            messageEl.style.color = 'var(--accent3)';
                            return;
                        }
                    }
                }

                const tri1Set = new Set(triangle1Name.split(''));
                const tri2Set = new Set(triangle2Name.split(''));
                const isValidTriangleName = (name, set) =>
                    new Set(name.split('')).size === 3 && [...set].every(l => name.includes(l));
                if (!isValidTriangleName(ansTri1, tri1Set) || !isValidTriangleName(ansTri2, tri2Set) ||
                    !isValidTriangleName(ansTri1b, tri1Set) || !isValidTriangleName(ansTri2b, tri2Set)) {
                    messageEl.innerHTML = t('invalid_triangle');
                    messageEl.style.color = 'var(--accent3)';
                    attempts++;
                    updateStatsUI();
                    if (attempts >= 10) finishCurrentRound();
                    else generateNewQuestion();
                    return;
                }

                const ang1Vertex = vertexFromAngle(ansAng1, tri1Vertices);
                const ang2Vertex = vertexFromAngle(ansAng2, tri2Vertices);

                if (!ang1Vertex || !ang2Vertex) {
                    messageEl.innerHTML = t('invalid_angle_format');
                    messageEl.style.color = 'var(--accent3)';
                    attempts++;
                    updateStatsUI();
                    if (attempts >= 10) finishCurrentRound();
                    else generateNewQuestion();
                    return;
                }

                const correctAngVertexTri1 = tri1Vertices[chosenVertexIndex];
                const mappedAngEntry = angleMapping.find(m => m.tri1Index === chosenVertexIndex);
                const correctAngVertexTri2 = tri2Vertices[mappedAngEntry.tri2Index];

                if (ang1Vertex !== correctAngVertexTri1 || ang2Vertex !== correctAngVertexTri2) {
                    messageEl.innerHTML = t('wrong_angle');
                    messageEl.style.color = 'var(--accent3)';
                    attempts++;
                    updateStatsUI();
                    if (attempts >= 10) finishCurrentRound();
                    else generateNewQuestion();
                    return;
                }

                // الضلعان في المثلث الأول
                const tri1SideSingleIdx = chosenVertexIndex;
                const tri1SideDoubleIdx = (chosenVertexIndex + 2) % 3;
                const tri1SideSingleA = tri1Vertices[tri1SideSingleIdx];
                const tri1SideSingleB = tri1Vertices[(tri1SideSingleIdx + 1) % 3];
                const tri1SideDoubleA = tri1Vertices[tri1SideDoubleIdx];
                const tri1SideDoubleB = tri1Vertices[chosenVertexIndex];

                // الضلعان في المثلث الثاني
                const tri2AngleVertexIdx = mappedAngEntry.tri2Index;
                const tri2SideSingleIdx = tri2AngleVertexIdx;
                const tri2SideDoubleIdx = (tri2AngleVertexIdx + 2) % 3;
                const tri2SideSingleA = tri2Vertices[tri2SideSingleIdx];
                const tri2SideSingleB = tri2Vertices[(tri2SideSingleIdx + 1) % 3];
                const tri2SideDoubleA = tri2Vertices[tri2SideDoubleIdx];
                const tri2SideDoubleB = tri2Vertices[tri2AngleVertexIdx];

                function sidePairMatches(s1, s2, labels1, idx1a, idx1b, labels2, idx2a, idx2b) {
                    const match1 = sideMatches(s1, labels1, idx1a, idx1b);
                    const match2 = sideMatches(s2, labels2, idx2a, idx2b);
                    return match1 && match2;
                }

                const pairAMatchesSingle = sidePairMatches(ansSideA1, ansSideA2,
                    tri1Vertices, tri1SideSingleIdx, (tri1SideSingleIdx + 1) % 3,
                    tri2Vertices, tri2SideSingleIdx, (tri2SideSingleIdx + 1) % 3);
                const pairAMatchesDouble = sidePairMatches(ansSideA1, ansSideA2,
                    tri1Vertices, tri1SideDoubleIdx, chosenVertexIndex,
                    tri2Vertices, tri2SideDoubleIdx, tri2AngleVertexIdx);
                const pairBMatchesSingle = sidePairMatches(ansSideB1, ansSideB2,
                    tri1Vertices, tri1SideSingleIdx, (tri1SideSingleIdx + 1) % 3,
                    tri2Vertices, tri2SideSingleIdx, (tri2SideSingleIdx + 1) % 3);
                const pairBMatchesDouble = sidePairMatches(ansSideB1, ansSideB2,
                    tri1Vertices, tri1SideDoubleIdx, chosenVertexIndex,
                    tri2Vertices, tri2SideDoubleIdx, tri2AngleVertexIdx);

                const validPairs = (pairAMatchesSingle && pairBMatchesDouble) || (pairAMatchesDouble &&
                    pairBMatchesSingle);

                if (!validPairs) {
                    messageEl.innerHTML = t('wrong_sides');
                    messageEl.style.color = 'var(--accent3)';
                    attempts++;
                    updateStatsUI();
                    if (attempts >= 10) finishCurrentRound();
                    else generateNewQuestion();
                    return;
                }

                const allSideInputs = [ansSideA1, ansSideA2, ansSideB1, ansSideB2];
                const allAngInputs = [ansAng1, ansAng2];
                const isCorrect = (new Set(allSideInputs)).size === 4 && (new Set(allAngInputs)).size === 2;

                attempts++;
                if (isCorrect) {
                    score++;
                    messageEl.innerHTML = t('correct');
                    messageEl.style.color = 'var(--accent2)';
                    playSound('correct');
                } else {
                    const correctAngName1 = getAngleNamesForVertex(chosenVertexIndex, tri1Vertices).standard;
                    const correctAngName2 = getAngleNamesForVertex(tri2AngleVertexIdx, tri2Vertices).standard;
                    const sideSLabel = tri1SideSingleA + tri1SideSingleB;
                    const sideS2Label = tri2SideSingleA + tri2SideSingleB;
                    const sideDLabel = tri1SideDoubleA + tri1SideDoubleB;
                    const sideD2Label = tri2SideDoubleA + tri2SideDoubleB;
                    messageEl.innerHTML = t('wrong', {
                        tri1: triangle1Name, tri2: triangle2Name,
                        ang1: correctAngName1, ang2: correctAngName2,
                        sideS1: sideSLabel, sideS2: sideS2Label,
                        sideD1: sideDLabel, sideD2: sideD2Label
                    });
                    messageEl.style.color = 'var(--accent3)';
                    playSound('incorrect');
                }

                updateStatsUI();
                if (attempts >= 10) {
                    finishCurrentRound();
                    return;
                }
                generateNewQuestion();
                setTimeout(() => inputRefs.tri1?.focus(), 40);
            }

            function onManualNewQuestion() {
                if (!gameActive) {
                    messageEl.innerHTML = t('round_restart');
                    return;
                }
                generateNewQuestion();
                messageEl.innerHTML = t('figure_changed');
                messageEl.style.color = 'var(--text)';
                setTimeout(() => {
                    if (messageEl.innerHTML === t('figure_changed')) messageEl.innerHTML = '';
                }, 1500);
                setTimeout(() => inputRefs.tri1?.focus(), 40);
            }

            // ══════════════ ربط الأحداث ══════════════
            checkBtn.addEventListener('click', handleCheck);
            generateBtn.addEventListener('click', onManualNewQuestion);
            questionContainer.addEventListener('keypress', (e) => {
                if (e.key === 'Enter' && e.target.classList.contains('inline-input')) {
                    e.preventDefault();
                    handleCheck();
                }
            });

            btnDarkMode.addEventListener('click', toggleTheme);
            btnSound.addEventListener('click', toggleSound);
            langSelect.addEventListener('change', (e) => applyLanguage(e.target.value));
            btnBack.addEventListener('click', goBack);

            // ══════════════ بدء التشغيل ══════════════
            loadLanguage();
            loadTheme();
            loadSoundPref();

            attempts = 0;
            score = 0;
            gameActive = true;
            isSwitchingRound = false;
            const initData = generateAllData();
            currentTri1Pts = initData.pts1;
            currentTri2Pts = initData.pts2;
            renderQuestion();
            drawFigures();
            updateStatsUI();
            messageEl.innerHTML = '';
            checkBtn.disabled = false;
            setTimeout(() => inputRefs.tri1?.focus(), 100);

            updateStaticI18n();
        })();
    </script>
